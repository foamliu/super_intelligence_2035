proc logicsynth_build_lint_scaffold {rtl_relpath module_name} {
    # design_rtl.list
    set dir [file dirname $rtl_relpath]
    set rl "+incdir+../../$dir\n../../$rtl_relpath\n"

    # prj
    set prj "#!SPYGLASS_PROJECT_FILE\n"
    append prj "#!VERSION 3.0\n"
    append prj "#  SpyGlass project file for $module_name (GUI complement to batch Makefile)\n"
    append prj "#  Usage: cd lint/work && spyglass -project $module_name.prj\n"
    append prj "\n"
    append prj "##Data Import Section\n\n"
    append prj "read_file -type sourcelist design_rtl.list\n\n"
    append prj "##Common Options Section\n\n"
    append prj "set_option projectwdir ./\n"
    append prj "set_option language_mode mixed\n"
    append prj "set_option designread_enable_synthesis no\n"
    append prj "set_option designread_disable_flatten no\n"
    append prj "set_option enableSV yes\n"
    append prj "set_option top $module_name\n"
    append prj "set_option active_methodology \$SPYGLASS_HOME/GuideWare/latest/block/rtl_handoff\n\n"
    append prj "##Goal Setup Section\n\n"
    append prj "current_methodology \$SPYGLASS_HOME/GuideWare/latest/block/rtl_handoff\n"

    # Makefile
    set mk "#==============================================================\n"
    append mk "# $module_name Digital Static Check (Lint / CDC / RDC)\n"
    append mk "# Usage (in work/):\n"
    append mk "#   make lint    -- RTL lint check     -> ../report/lint.rpt\n"
    append mk "#   make cdc     -- CDC check          -> ../report/cdc.rpt\n"
    append mk "#   make rdc     -- RDC check          -> ../report/rdc.rpt\n"
    append mk "#   make all     -- lint + cdc + rdc\n"
    append mk "#   make clean   -- remove generated reports/logs\n"
    append mk "#==============================================================\n\n"
    append mk "# project\n"
    append mk "PROJECT_ROOT = /proj\n"
    append mk "PROJECT_NAME = lagat\n"
    append mk "USER = \$(whoami)\n\n"
    append mk "# tool (VC Static / SpyGlass static-check platform)\n"
    append mk "TOOL = ds/vc_static\n"
    append mk "VERSION = V-2023.12-SP1\n\n"
    append mk "# lsf\n"
    append mk "QUEUE = fqueue2h\n\n"
    append mk "# design\n"
    append mk "DESIGN_TOP = $module_name\n"
    append mk "WORKDIR = workdir\n"
    append mk "INCDIR = ./\n"
    append mk "LIB_FILE =\n"
    append mk "LIBEXT = .v\n"
    append mk "SEARCH_PATH = .\n"
    append mk "LINK_LIBRARY =\n"
    append mk "ENABLE_SV_PARSER = 1\n\n"
    append mk "# lint settings\n"
    append mk "LINT_GOAL = lint_rtl\n"
    append mk "LINT_BLACKBOX_MODULE =\n"
    append mk "LINT_WAIVER = ../config/waiver.tcl\n\n"
    append mk "# cdc settings\n"
    append mk "CDC_SDC  = ../config/cdc_sdc/cdc.sdc\n"
    append mk "CDC_SDC +=\n\n"
    append mk "# rdc settings\n"
    append mk "RDC_SDC  = ../config/rdc_sdc/rdc.sdc\n"
    append mk "RDC_SDC +=\n\n"
    append mk "# filename settings\n"
    append mk "LINT_TCL_FILE = lint.tcl\n"
    append mk "CDC_TCL_FILE  = cdc.tcl\n"
    append mk "RDC_TCL_FILE  = rdc.tcl\n"
    append mk "LINT_REPORT_FILE = ../report/lint.rpt\n"
    append mk "CDC_REPORT_FILE  = ../report/cdc.rpt\n"
    append mk "RDC_REPORT_FILE  = ../report/rdc.rpt\n\n"
    append mk "clean:\n"
    append mk "\trm -rf workdir/*.rpt workdir/*.log\n\n"
    append mk "include /apps/imctf/cad/script/digital_static_check/sc_target.mk\n"

    return [dict create design_rtl_list $rl prj $prj makefile $mk]
}
